TAD AirMiles {

    Categoria ES enum {Bronce, Plata, Oro, Platino}

    (STRUCTS AUX):{

    promocion ES struct<
        factor: R
        tope: R
        gastadas: R
    >

    vuelo ES struct <
        distancia: R
        millasGanadas: R 
        promocion: String 
        factor: R
    >

    transaccion ES struct<
        identificador_usuario2:Z 
        monto: Z
    >


    usuario ES struct <
        millasTotales: R
        millasDisponibles: R
        millasCanjeadas: R 
        historialVuelos: seq<vuelo>
        historialTransacciones: seq <transaccion>
    >


    obs usuarios: dicc <Id,usuario>
    obs promociones: dicc <nombrePromocion,promocion>
    }

    pred estaRegistrado(d:dicc<id,usuario>,id_usuario:Z) {id_usuario pertenece d }


    crearSistema(): AirMiles // 
        requiere: {true}
        asegura: {res.usuarios = {}}
        asegura: {res.promociones = setKey(res.promociones,'NoPromo', (1,0,0) )} // (factor, tope, gastadas)
        

    //listo
    registrarUsuario(inout a: AirMiles, in id_usuario: Z): // noo, solo en este proc y en el regristrarUsuarioConReferidor
        requiere: {a0 = a}
        requiere: {not estaRegistrado(a.usuarios, id_usuario)}
        requiere: {id_usuario > 0} 
        asegura:  {a.usuarios = setKey (a0.usuarios, id_usuario,(0,0,0,<>,<>))  }   
        asegura:  {a.promociones = a0.promociones}       // (totales,disp,canjeadas,histvuelos,histTransacc)                                                                                  


    //faltaArreglarCategoría  /->L ó /<-> /noHAceFaltaPonerLaCategoriaEnformaDeStrignSInoDEFORMA:a0.usuarios[referidor].millasTotales>=1.000.000 (Esto ya garantiza que es platino no?)
    regristrarUsuarioConReferidor(inout a: AirMiles, in nuevo_usuario: Z, in referidor: Z)
        requiere: {a0 = a, a1 = a}
        requiere: {(not estaRegistrado(a.usuarios, nuevo_usuario) and estaRegistrado(a.usuarios, referidor))}
        requiere: {nuevo_usuario > 0}     
        asegura:  {          a0.usuarios[referidor] es platino          ->L a.usuarios = setKey(a0.usuarios, referidor, (a0.usuario[referidor].millasTotales, a0.usuario[referidor].millasDisponibles + 500, a0.usuario[referidor].historialVuelos,a0.usuario[referidor].historialTransacciones))  y luego a1 = a}
        asegura:  {     not  a0.usuarios[referidor] es platino          ->L a.usuarios = setKey(a0.usuarios, referidor, (a0.usuario[referidor].millasTotales + 500, a0.usuario[referidor].millasDisponibles + 500, a0.usuario[referidor].historialVuelos,a0.usuario[referidor].historialTransacciones)) y luego a1 = a}
        asegura:  {a.usuarios = setKey(a1.usuarios, nuevo_usuario, usuario.millasTotales = 500 and usuario.millasDisponibles = 500 and usuario.millasCanjeadas = 0 and usuario.historialVuelos = () and usuario.historialTransacciones = ())}
        asegura:  {a.promociones = a0.promociones}   




    acumularMillas(inout a: AirMiles, in id_usuario: Z, in distancia: R, in nombrePromo: String)
        requiere: {a0 = a}
        requiere: {estaRegistrado(a.usuarios, id_usuario)}
        requiere: {distancia > 0}
        requiere: {nombrePromo in claves(a0.promociones) and (a0.promociones.tope > 0 ->L millasGanadasEnVuelo(a0, id_usuario, distancia, nombrePromo) < a0.promociones[nombrePromo].tope)} // aca chequeo que no se pase del tope cuando es != NoPromo
        //arreglar asegura: {a.usuarios = setKey(a0.usuarios, id_usuario, (usuario.millasDisponibles + millasGanadasEnVuelo(a0, id_usuario, distancia, nombrePromo),usuario.millasTotales + millasGanadasEnVuelo(a0, id_usuario, distancia, nombrePromo),usuario.historialVuelos ++ <(distancia, id_usuario, millasGanadasEnVuelo(a0, id_usuario, distancia, nombrePromo))>, nombrePromo, a0.promociones[nombrePromo].factor))}
        asegura: {a.promociones = ifThenElse(a0.promociones.tope = a0.promociones.gastadas + millasGanadasEnVuelo(a0, id_usuario, distancia, nombrePromo), delKey(a0.promociones,nombrePromo), a.promociones = setKey(a0.promociones, nombrePromo, gastadas + millasGanadasEnVuelo(a0, id_usuario, distancia, nombrePromo)))}
    
    //faltaArreglarCategoría
    // aux millasGanadasEnVuelo(a: AirMiles, id_usuario: Z, distancia: R, nombrePromo: String): R = a0.categoria{consultarCategoria(id_usuario,a)} * distancia * a.promociones[nombrePromo].factor

    //listo
    canjearMillas(inout a: AirMiles, in id_usuario: Z, in millas: R): // chequear si está bien
        requiere: {a0 = a}
        requiere: {estaRegistrado(a.usuarios, id_usuario)}
        requiere: {a.usuarios[id_usuario].millasDisponibles >= millas}
        asegura:  {a.usuarios = setKey(a0.usuarios, id_usuario, (a0.usuario[id_usuario].totales, a0.usuario[id_usuario].millasDisponibles - millas, a0.usuario[id_usuario].millasCanjeadas + millas, a0.usuario[referidor].historialVuelos,a0.usuario[referidor].historialTransacciones))}
        asegura:  {a.promociones = a0.promociones} 

    //categoria,metavariable en asegura?
    transferirMillas(inout a: AirMiles, in Da: Z, in Recibe: Z, in millas: R): // chequear si está bien
        requiere: {a0 = a, a1 = a}
        requiere: {Da != Recibe}
        requiere: {estaRegistrado(a.usuarios, Da) and estaRegistrado(a.usuarios, Recibe)}
        requiere {a0.usuarios[Da].millasDisponibles >= millas}
        asegura: {a = setKey(a0.usuarios, Da, (a0.usuario[Da].totales, a0.usuario[Da].millasDisponibles - millas, a0.usuario[Da].millasCanjeadas, a0.usuario[Da].historialVuelos, a0.usuario[Da].historial_transacciones ++ <(Recibe,monto*-1)>)) y luego a1 = a}
        //asegura: {a = setKey(a0.usuarios, Da, (a0.usuario[Da].totales, a0.usuario[Da].millasDisponibles - millas, a0.usuario[Da].millasCanjeadas, a0.usuario[Da].historialVuelos, a0.usuario[Da].historial_transacciones ++ <(Recibe,monto*-1)>))}
        asegura: {a.promociones = a0.promociones} 

    //listo
    consultarSaldo(in a: AirMiles, in id_usuario: Z): R
        requiere: {estaRegistrado(a.usuarios, Z)}
        asegura: {res = a.usuarios[id_usuario].millasDisponibles}

    consultarCategoria (in a: AirMiles, in id_usuario: Z): Categoria
        requiere: {estaRegistrado(a.usuarios, id_usuario)}
        asegura: {esBronce(a.usuarios[id_usuario].millasTotales) -> res = Categoria.Bronce}
        asegura: {esPlata(a.usuarios[id_usuario].millasTotales) -> res = Categoria.Plata}
        asegura: {esOro(a.usuarios[id_usuario].millasTotales) -> res = Categoria.Oro} 
        asegura: {esPlatino(a.usuarios[id_usuario].millasTotales) -> res = Categoria.Platino}

    pred esBronce(millas: Z) {millas < 10000}
    pred esPlata(millas: Z) {millas >= 10000 and millas <= 49999}
    pred esOro(millas: Z) {millas >= 50000 and millas <= 999999}
    pred esPlatino(millas: Z) {millas >= 1000000}

    //listo
    rankingMillas (in a: AirMiles, in cant_ids: Z): seq<Id> // problema con rango de i
        requiere: {cant_ids > 0}
        asegura: {ifThenElse(Z > |a.usuarios|, |res| = |a.usuarios|, |res| = cant_ids) y PerteneceSeq(res,claves(a.usuario)) }
        asegura: {para todo 0 <= i < |res|-1 ->luego a.usuarios[res[i]].millasDisponibles >= a.usuarios[res[i+1]].millasDisponibles}
        asegura: {(para todo i : Id) i E a.usuarios Y i !E res ->L res[|res|-1] > a.usuarios[i]} //se fija que sean los mejores


    pred PerteneceSeq(s: seq<Id>, keys: seq<Id>) {para todo 0 <= i < |s| ->luego s[i] E keys} // E = Pertenece

    resumenTransacciones(in a: AirMiles, int id_usuario: Z): tuple(R,R,R)
        requiere: {estaRegistrado(a.usuarios, id_usuario)}
        asegura: {res = (totalMillasPorViaje(a.usuario[id_usuario].historialVuelos), a.usuario[id_usuario].millasCanjeadas,millasTransferidasGeneral(a.usuarios[id_usuario].historial_transacciones))}

    aux millasTransferidasGeneral(in t seq<transaccion>): R = sumatoria de i = 0 hasta i = |t|-1 de t[i].monto
    aux totalMillasPorViaje(in v seq<vuelos>): R = sumatoria de i = 0 hasta i = |v|-1 de v[i].millasGanadas


    iniciarPromocion(inout a: AirMiles, in nombre: String, in factor: R):
        requiere: {a0 = a}
        requiere: {factor >= 1}
        requiere: {nombre not in claves(a0.promociones)}
        asegura:  {a.promociones = setKey(a0.promociones, nombre, (factor,0,0))}
        asegura:  {a.usuarios = a0.usuarios}

    iniciarPromocionConTope(inout a: AirMiles, in nombre: String, in factor: R, in tope: R):
        requiere: {a0 = a}
        requiere: {factor >= 1}
        requiere: {tope > 0}
        requiere: {nombre not in claves(a0.promociones)}
        asegura:  {a.promociones = setKey(a0.promociones, nombre, (factor,tope,0))}
        asegura:  {a.usuarios = a0.usuarios}

    finalizarPromocion(inout a: AirMiles, in nombre: String):
        requiere: {a0 = a}
        requiere: {nombre in claves(a0.promociones) and nombre != "NoPromo"}
        asegura:  {a.promociones = delKey(a0,nombre)}
        asegura:  {a.usuarios = a0.usuarios}

    elQueTieneMasMillas(in a: AirMiles): Id
        requiere: {|a.usuarios| > 0}
        asegura:  {(para toda key de tipo id), key in claves(a.usuarios) ->luego a.usuarios[res].millasTotales >= a.usuarios[key].millasTotales}

    elMasCanjeador(in a: AirMiles): Id
        requiere: {|a.usuarios| > 0}
        asegura: {(para toda key de tipo id), key in claves(a.usuarios) ->luego (a.usuarios[res].millasCanjeadas + cantidadMillasDadas(a.usuarios[res].historial_transacciones)) >= (a.usuarios[key].millasCanjeadas + cantidadMillasDadas(a.usuarios[key].historial_transacciones))}

    aux cantidadMillasDadas(in transferencias: seq<transaccion>) : R = -1* (sumatoria de i = 0 hasta i = |v|-1 ifThenElse(transferencias[i].monto > 0,0,transferencias[i].monto))
    
    

    paresTransferidoresExclusivos(in a: AirMiles): seq<tuple(Id,Id)>
        requiere: {true}
        asegura: {NoHayTuplasRepetidas(res) y transferenciasExclusivas(a.usuarios,res) y estaRegistrado(a.usuarios,res) }
    
    pred estaRegistrado (usuario: dicc <Id,usuario> , ids:seq<struct(id1:Z,id2:Z)>): {
        (v i : z)(0<=i|ids| ->l  estaRegistrado(usuario, ids[i].id1) y estaRegistrado(usuario,ids[i].id2) )
    }

    pred NoHayTuplasRepetidas(s:seq<struct(id1:Z,id2:Z)>){
        (vi:z)(0<=i<|s| ->luego not (Existe j:z)(0<=j<|s| y j/=i ->L (i.id1=j.id1 o i.id1 = j.id2 Y i.id2=j.id1 o i.id2=j.id2) )

        //noSe //siAgregarAlgoMAs
    }

    pred transferenciasExclusivas(usuario: dicc <Id,usuario>, s: seq<struct(id1:Z,id2:Z)>){
        (v i : z)(0<= i < |s| ->l (Existe j:z)(0<=j < |usuario[i.id1].historialTransacciones| yluego ( j.identificador_usuario2=i.id2))
        o
        (v i : z)(0<= i < |s| ->l (Existe j:z)(0<=j < |usuario[i.id2].historialTransacciones| yluego ( j.identificador_usuario2=i.id1))
    }

}
